-- Crear tipo de rol de usuario
DO $$
BEGIN
	IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'user_role') THEN
		CREATE TYPE user_role AS ENUM ('administrador', 'comun', 'profesional');
	END IF;
END
$$;

DO $$
BEGIN
	IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'usuario') THEN
		CREATE USER usuario WITH PASSWORD 'contrasena';
	END IF;
END
$$;

GRANT CONNECT ON DATABASE mi_proyecto TO usuario;
GRANT USAGE, CREATE ON SCHEMA public TO usuario;

    id SERIAL PRIMARY KEY,
    nombre_usuario VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,

);

CREATE TABLE IF NOT EXISTS bloque_horario (

    id SERIAL PRIMARY KEY,
    id_reserva INTEGER NOT NULL,
    id_bloque INTEGER NOT NULL,

CREATE OR REPLACE FUNCTION actualizar_estado_componente()
RETURNS TRIGGER AS $$
BEGIN

    -- Sin stock
    IF NEW.cantidad <= 0 THEN
        NEW.estado := 'agotado';

    -- Stock crítico
    ELSIF NEW.cantidad <= NEW.stock_critico THEN
        NEW.estado := 'critico';

    -- Stock normal
    ELSE
        NEW.estado := 'disponible';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
